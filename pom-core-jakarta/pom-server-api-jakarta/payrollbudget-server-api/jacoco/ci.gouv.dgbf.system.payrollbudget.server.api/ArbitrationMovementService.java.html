/**
 * Contrat de service des lignes d'arbitrage.
 *
 * <p>Ce service n'expose <strong>aucune suppression</strong>. Une ligne d'arbitrage ne
 * disparaît jamais : c'est ce qui rend l'arbitrage auditable.</p>
 *
 * <p>Il expose en revanche une <strong>création</strong>, et c'est la spécification qui
 * l'impose : {@code CU03-RG02}, {@code CU04-RG02} et {@code CU05-RG02} disent, pour chacune des
 * trois familles, que tout élément <em>transmis au cours des traitements</em> est pris en
 * compte dans le besoin exprimé par l'EPN. Le besoin n'est donc pas clos à la transmission :
 * ce qui arrive ensuite à la DCB — par courrier, en séance, en réponse à une demande — y
 * entre aussi. Sans cette porte, ces éléments resteraient hors du système et l'arbitrage
 * porterait sur un besoin incomplet.</p>
 *
 * <p>Créer n'est pas modifier. {@code CU03-RG01}, {@code CU04-RG01} et {@code CU05-RG01}
 * interdisent au chargé de toucher aux besoins exprimés, qu'il ne fait que consulter : une
 * ligne venue de l'EPN reste intacte, quoi qu'il arrive. C'est aussi la raison pour laquelle
 * l'ajustement se saisit comme un <em>montant à ajuster</em>, posé à côté du montant de
 * l'EPN, et jamais en réécrivant celui-ci.</p>
 *
 * @author Christian
 *
 */
@Path(value = ArbitrationMovementService.PATH)
@Tag(name = "Gestion des lignes d'arbitrage")

  /**
   * Ajoute au dossier un élément parvenu à la DCB après la transmission
   * (CU03-RG02, CU04-RG02, CU05-RG02).
   *
   * <p>La ligne naît <strong>admise</strong> : la DCB ne l'ajoute qu'en ayant sous les yeux ce
   * qui la justifie, et lui faire reparcourir le visa, la transmission et la réception serait
   * rejouer une histoire qui n'a pas eu lieu. Elle n'est pas pour autant accordée — le
   * chargé la traite comme les autres, et sa décision est tracée comme les autres.</p>
   *
   * <p>Elle ne porte aucune ligne <em>réelle</em> d'expression des besoins : sa référence est
   * forgée, et son préfixe {@link ArbitrationMovementDto#EXPRESSION_LINE_DCB_PREFIX} l'annonce.
   * C'est à cela qu'on la reconnaît, pour toujours, comme un ajout de la DCB et non une
   * demande de l'EPN.</p>
   *
   * @param request l'élément à ajouter
   * @return l'identifiant attribué
   */
  @Path(CREATE_PATH)
  @POST
  @Consumes(MediaType.APPLICATION_JSON)
  @Produces(value = {MediaType.APPLICATION_JSON, MediaType.TEXT_PLAIN})
  @Operation(operationId = CREATE_IDENTIFIER,
      summary = "Ajout d'un élément au cours du traitement")
  @APIResponse(responseCode = "201",
      content = {@Content(schema = @Schema(implementation = CreateResponseDto.class))})
  Response create(ArbitrationMovementCreateRequestDto request);

  /**
   * Requête d'ajout d'un élément au cours du traitement.
   *
   * <p>Le montant d'impact est <strong>saisi</strong>, seul cas du module où il l'est : un
   * élément qui n'est jamais passé par l'expression des besoins n'a pas de Δ calculé dont on
   * pourrait le déduire. Il se motive donc, comme tout ce qui ne se déduit pas.</p>
   */
  @Getter
  @Setter
  class ArbitrationMovementCreateRequestDto extends AbstractAuditedRequestJsonDto
      implements ArbitrationMovementSaveRequestDto {

    /**
     * Dossier auquel la ligne est ajoutée.
     */
    @JsonbProperty(JSON_DOSSIER_IDENTIFIER)
    private String dossierIdentifier;

    /**
     * Nature de l'élément ajouté : mouvement, mesure ponctuelle ou transfert de solde.
     */
    @JsonbProperty(JSON_ELEMENT_TYPE)
    private String elementType;

    /**
     * Code du type de mouvement, qui dit aussi quel droit l'instruit.
     */
    @JsonbProperty(JSON_MOVEMENT_TYPE_CODE)
    private String movementTypeCode;

    /**
     * Agent concerné, vide pour une ligne qui n'en désigne aucun.
     */
    @JsonbProperty(JSON_EMPLOYEE_IDENTIFIER)
    private String employeeIdentifier;

    /**
     * Unité administrative d'origine, renseignée pour un transfert de solde.
     */
    @JsonbProperty(JSON_ORIGIN_ADMINISTRATIVE_UNIT_IDENTIFIER)
    private String originAdministrativeUnitIdentifier;

    /**
     * Montant que la ligne pèse sur la masse salariale.
     */
    @JsonbProperty(JSON_IMPACT_AMOUNT)
    private Long impactAmount;

    /**
     * Pourquoi la DCB ajoute cet élément. Obligatoire.
     */
    @JsonbProperty(JSON_REASON)
    private String reason;
  }

  /**
   * Renvoie une ligne au traitement : le chef d'équipe n'a pas retenu le travail rendu.
   *
   * <p>Le chef vient confirmer ou rejeter ce que son équipe a traité. Confirmer menait déjà
   * quelque part — la validation de la ligne ; <strong>rejeter ne menait nulle part</strong>, et
   * son désaccord n'avait aucune forme : il lui restait à ne pas valider, ce qui bloque le
   * dossier sans rien dire à personne.</p>
   *
   * <p>La ligne revient donc à <em>Réceptionné</em>, c'est à dire dans la liste de l'agent, avec
   * le motif du chef. Elle n'est pas annulée : elle est à refaire.</p>
   *
   * <p><strong>Seule une acceptation revient.</strong> Le rejet prononcé au traitement est
   * définitif, et cette règle n'est pas revue ici : la ligne rejetée est sortie du circuit, aucune
   * fermeture ne l'attend, et la consolidation ne la compte pas. La rendre réversible toucherait
   * aux trois à la fois, et c'est une autre décision que celle-ci.</p>
   *
   * <p><strong>Le motif est obligatoire.</strong> Un travail renvoyé sans motif est un travail
   * que l'agent refera à l'identique.</p>
   *
   * @param request identifiant de la ligne et motif du renvoi
   * @return l'identifiant de la ligne touchée
   */
  @Path(RETURN_TO_PROCESSING_PATH)
  @POST
  @Consumes(MediaType.APPLICATION_JSON)
  @Produces(value = {MediaType.APPLICATION_JSON, MediaType.TEXT_PLAIN})
  @Operation(operationId = RETURN_TO_PROCESSING_IDENTIFIER,
      summary = "Renvoi d'une ligne d'arbitrage au traitement")
  @APIResponse(responseCode = "200",
      content = {@Content(schema = @Schema(implementation = IdentifiableResponseDto.class))})
  Response returnToProcessing(ArbitrationMovementTransitionRequestDto request);

  /**

   * <p>Trois chemins y mènent, et ils n'ont pas le même sens :</p>
   * <ul>
   *   <li>depuis la <em>réception</em>, réservée aux lignes d'initialisation, réputées déjà
   *       arbitrées (RG_PRB_ARB_17) ;</li>
   *   <li>depuis <em>Validé</em>, c'est la consolidation proprement dite : parmi ce que
   *       l'arbitrage a retenu et validé, on choisit ce qui entre dans la masse finale ;</li>
   *   <li>depuis <em>Non consolidé</em>, c'est la reprise d'une ligne mise à l'écart.</li>
   * </ul>
   *
   * @param request identifiant de la ligne
   * @return l'identifiant de la ligne touchée

  String NOT_CONSOLIDATE_PATH = "non-consolidation";

  /**
   * Met une ligne à l'écart de la masse salariale finale.
   *
   * <p>Ce n'est <strong>pas un rejet</strong>, et c'est toute la différence. Le rejet dit que la
   * demande n'était pas fondée ; la non-consolidation dit qu'elle l'était, mais que l'exercice
   * ne la portera pas — le plus souvent faute d'enveloppe. La ligne sort du cumul sans que
   * l'arbitrage rendu sur elle soit défait.</p>
   *
   * <p><strong>Elle peut revenir.</strong> {@code consolidate} la ramène dans la masse, et son
   * montant avec elle. C'est ce qui distingue la mise à l'écart du rejet de traitement, lui
   * définitif (RG_PRB_ARB_22), et c'est ce qui tranche le point ouvert O7 du modèle d'états du
   * 27/08/2026 — lequel demandait à confirmer si « Non consolidé » était vraiment terminal.
   * Il ne l'est pas.</p>
   *
   * <p>Le motif est obligatoire : une ligne arbitrée favorablement puis écartée sans explication
   * serait indistinguable d'un oubli, et rien ne dirait à quelle condition la reprendre.</p>
   *
   * @param request identifiant de la ligne et motif
   * @return l'identifiant de la ligne touchée
   */

      summary = "Mettre une ligne d'arbitrage à l'écart de la masse salariale")
